cmake_minimum_required(VERSION 3.16)
project(poplan_native VERSION 0.1.0 LANGUAGES CXX)
option(POPLAN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(poplan_runtime src/value.cpp src/compiler.cpp src/runtime.cpp src/primitives.cpp)
target_include_directories(poplan_runtime PUBLIC include)
target_compile_features(poplan_runtime PUBLIC cxx_std_17)
target_compile_options(poplan_runtime PRIVATE -Wall -Wextra -Wpedantic)
if(POPLAN_SANITIZERS)
  target_compile_options(poplan_runtime PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(poplan_runtime PUBLIC -fsanitize=address,undefined)
endif()
add_executable(poplan-native src/main.cpp)
target_link_libraries(poplan-native PRIVATE poplan_runtime)
file(COPY lib DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME standalone-interactive COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/interactive.py $<TARGET_FILE:poplan-native>)
  add_test(NAME standalone-conformance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/conformance.py $<TARGET_FILE:poplan-native>)
endif()
install(TARGETS poplan-native RUNTIME DESTINATION bin)
install(DIRECTORY lib/ DESTINATION share/poplan/lib)
if(BUILD_TESTING)
  add_executable(standalone-runtime-test tests/runtime_test.cpp)
  target_link_libraries(standalone-runtime-test PRIVATE poplan_runtime)
  add_test(NAME standalone-runtime COMMAND standalone-runtime-test)
endif()
if(BUILD_TESTING)
  add_executable(standalone-session-test tests/session_test.cpp)
  target_link_libraries(standalone-session-test PRIVATE poplan_runtime)
  add_test(NAME standalone-session COMMAND standalone-session-test)
endif()
